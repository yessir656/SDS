   277→   277→interface FormState {

   299→   299→
   300→   300→function ChemicalFormDialog({
   301→   301→  chemical,
   302→   302→  onClose,
   303→   303→  onSaved,
   304→   304→}: {
   305→   305→  chemical: AdminChemical | null;
   306→   306→  onClose: () => void;
   307→   307→  onSaved: () => void;
   308→   308→}) {
   309→   309→  const [form, setForm] = useState<FormState>({
   310→   310→    id: chemical?.id ?? "",
   311→   311→    casNumber: chemical?.casNumber ?? "",
   312→   312→    chemicalName: chemical?.chemicalName ?? "",
   313→   313→    formula: chemical?.formula ?? "",
   314→   314→    tradeName: chemical?.tradeName ?? "",
   315→   315→    manufacturer: chemical?.manufacturer ?? "",
   316→   316→    supplier: chemical?.supplier ?? "",
   317→   317→    signalWord: (chemical?.signalWord as SignalWord) ?? "danger",
   318→   318→    hazardClasses: (chemical?.hazardClasses as HazardClass[]) ?? [],
   319→   319→    ghsPictograms: (chemical?.ghsPictograms as GhsPictogram[]) ?? [],
   320→   320→    storageLocation: chemical?.storageLocation ?? "",
   321→   321→    department: (chemical?.department as Department) ?? "Chemical Analysis",
   322→   322→    safetyInstructions: chemical?.safetyInstructions ?? "",
   323→   323→    version: chemical?.version ?? "1.0",
   324→   324→    emergencyContact: chemical?.emergencyContact ?? "",
   325→   325→    personalProtectiveEquipment: chemical?.personalProtectiveEquipment.join("\n") ?? "",
   326→   326→    regulatoryTags: chemical?.regulatoryTags ?? [],
   327→   327→    firstAidMeasures: chemical?.firstAidMeasures ?? "",
   328→   328→    firefightingMeasures: chemical?.firefightingMeasures ?? "",
   329→   329→    accidentalReleaseMeasures: chemical?.accidentalReleaseMeasures ?? "",
   330→   330→  });
   331→   331→  const [saving, setSaving] = useState(false);
   332→   332→  const [error, setError] = useState<string | null>(null);
   333→   333→
   334→   334→  // AI auto-fill state.
   335→   335→  const [extracting, setExtracting] = useState(false);
   336→   336→  const [extractError, setExtractError] = useState<string | null>(null);
   337→   337→  const [extractedFromPdf, setExtractedFromPdf] = useState(false);
   338→   338→  const fileInputRef = useRef<HTMLInputElement>(null);
   339→   339→
   340→   340→  const isEdit = !!chemical;
   341→   341→
   342→   342→  // ---------------------------------------------------------------------------
   343→   343→  // AI auto-fill handler.
   344→   344→  // Sends the selected PDF to /api/admin/sds/extract, then merges the returned
   345→   345→  // fields into the form state. The `id` field is preserved (admin must enter
   346→   346→  // it manually in create mode).
   347→   347→  // ---------------------------------------------------------------------------
   348→   348→  const handleAutoFill = async (e: React.ChangeEvent<HTMLInputElement>) => {
   349→   349→    const file = e.target.files?.[0];
   350→   350→    // Reset the input value so the same file can be re-selected later.
   351→   351→    if (fileInputRef.current) fileInputRef.current.value = "";
   352→   352→    if (!file) return;
   353→   353→
   354→   354→    setExtracting(true);
   355→   355→    setExtractError(null);
   356→   356→
   357→   357→    try {
   358→   358→      const fd = new FormData();
   359→   359→      fd.append("file", file);
   360→   360→
   361→   361→      const res = await fetch("/api/admin/sds/extract", {
   362→   362→        method: "POST",
   363→   363→        body: fd,
   364→   364→      });
   365→   365→
   366→   366→      const json = await res.json().catch(() => ({ success: false, error: "Invalid response from server" }));
   367→   367→
   368→   368→      if (!res.ok || !json.success) {
   369→   369→        throw new Error(json.error || `Extraction failed (HTTP ${res.status})`);
   370→   370→      }
   371→   371→
   372→   372→      const d = json.data as {
   373→   373→        chemicalName?: string;
   374→   374→        casNumber?: string;
   375→   375→        formula?: string;
   376→   376→        tradeName?: string;
   377→   377→        manufacturer?: string;
   378→   378→        supplier?: string;
   379→   379→        signalWord?: SignalWord;
   380→   380→        ghsPictograms?: GhsPictogram[];
   381→   381→        hazardClasses?: HazardClass[];
   382→   382→        storageLocation?: string;
   383→   383→        safetyInstructions?: string;
   384→   384→        emergencyContact?: string;
   385→   385→        personalProtectiveEquipment?: string[];
   386→   386→        firstAidMeasures?: string;
   387→   387→        firefightingMeasures?: string;
   388→   388→        accidentalReleaseMeasures?: string;
   389→   389→      };
   390→   390→
   391→   391→      setForm((prev) => ({
   392→   392→        ...prev,
   393→   393→        // Preserve `id` — admin must always enter it manually in create mode.
   394→   394→        id: prev.id,
   395→   395→        chemicalName: d.chemicalName || prev.chemicalName,
   396→   396→        casNumber: d.casNumber || prev.casNumber,
   397→   397→        formula: d.formula || prev.formula,
   398→   398→        tradeName: d.tradeName || prev.tradeName,
   399→   399→        manufacturer: d.manufacturer || prev.manufacturer,
   400→   400→        supplier: d.supplier || prev.supplier,
   401→   401→        signalWord: d.signalWord || prev.signalWord,
   402→   402→        ghsPictograms:
   403→   403→          Array.isArray(d.ghsPictograms) && d.ghsPictograms.length > 0
   404→   404→            ? d.ghsPictograms
   405→   405→            : prev.ghsPictograms,
   406→   406→        hazardClasses:
   407→   407→          Array.isArray(d.hazardClasses) && d.hazardClasses.length > 0
   408→   408→            ? d.hazardClasses
   409→   409→            : prev.hazardClasses,
   410→   410→        storageLocation: d.storageLocation || prev.storageLocation,
   411→   411→        safetyInstructions: d.safetyInstructions || prev.safetyInstructions,
   412→   412→        emergencyContact: d.emergencyContact || prev.emergencyContact,
   413→   413→        // PPE comes back as an array; join with newlines for the textarea.
   414→   414→        personalProtectiveEquipment:
   415→   415→          Array.isArray(d.personalProtectiveEquipment) && d.personalProtectiveEquipment.length > 0
   416→   416→            ? d.personalProtectiveEquipment.join("\n")
   417→   417→            : prev.personalProtectiveEquipment,
   418→   418→        firstAidMeasures: d.firstAidMeasures || prev.firstAidMeasures,
   419→   419→        firefightingMeasures: d.firefightingMeasures || prev.firefightingMeasures,
   420→   420→        accidentalReleaseMeasures:
   421→   421→          d.accidentalReleaseMeasures || prev.accidentalReleaseMeasures,
   422→   422→      }));
   423→   423→
   424→   424→      setExtractedFromPdf(true);
   425→   425→    } catch (err) {
   426→   426→      setExtractError(err instanceof Error ? err.message : "Auto-fill failed");
   427→   427→      setExtractedFromPdf(false);
   428→   428→    } finally {
   429→   429→      setExtracting(false);
   430→   430→    }
   431→   431→  };
   432→   432→
   433→   433→  const handleSubmit = async (e: React.FormEvent) => {
   434→   434→    e.preventDefault();
   435→   435→    setSaving(true);
   436→   436→    setError(null);
   437→   437→
   438→   438→    try {
   439→   439→      const payload = {
   440→   440→        ...form,
   441→   441→        personalProtectiveEquipment: form.personalProtectiveEquipment
   442→   442→          .split("\n")
   443→   443→          .map((s) => s.trim())
   444→   444→          .filter(Boolean),
   445→   445→      };
   446→   446→
   447→   447→      const url = isEdit ? `/api/admin/chemicals/${form.id}` : "/api/admin/chemicals";
   448→   448→      const method = isEdit ? "PUT" : "POST";
   449→   449→
   450→   450→      const res = await fetch(url, {
   451→   451→        method,
   452→   452→        headers: { "Content-Type": "application/json" },
   453→   453→        body: JSON.stringify(payload),
   454→   454→      });
   455→   455→
   456→   456→      if (!res.ok) {
   457→   457→        const err = await res.json().catch(() => ({ error: "Request failed" }));
   458→   458→        throw new Error(err.error || `HTTP ${res.status}`);
   459→   459→      }
   460→   460→
   461→   461→      onSaved();
   462→   462→    } catch (err) {
   463→   463→      setError(err instanceof Error ? err.message : "Save failed");
   464→   464→    } finally {
   465→   465→      setSaving(false);
   466→   466→    }
   467→   467→  };
   468→   468→
   469→   469→  const toggleArray = <T,>(arr: T[], value: T): T[] => {
   470→   470→    return arr.includes(value) ? arr.filter((x) => x !== value) : [...arr, value];
   471→   471→  };
   472→   472→
   473→   473→  return (
