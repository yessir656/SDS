    82→
    83→function formatRelative(ts: number | null): string {
    84→  if (!ts) return "never";
    85→  const diff = Date.now() - ts;
    86→  const mins = Math.floor(diff / 60_000);
    87→  if (mins < 1) return "just now";
    88→  if (mins < 60) return `${mins}m ago`;
    89→  const hours = Math.floor(mins / 60);
    90→  if (hours < 24) return `${hours}h ago`;
    91→  const days = Math.floor(hours / 24);
    92→  if (days < 30) return `${days}d ago`;
    93→  return new Date(ts).toLocaleDateString();
    94→}

   389→              Delete permanently

   391→          </AlertDialogFooter>

   402→function CreateUserDialog({
   403→  open,
   404→  onOpenChange,
   405→  onCreated,
   406→}: {
   407→  open: boolean;
   408→  onOpenChange: (open: boolean) => void;
   409→  onCreated: () => void;
   410→}) {
   411→  const [email, setEmail] = useState("");
   412→  const [name, setName] = useState("");
   413→  const [password, setPassword] = useState("");
   414→  const [role, setRole] = useState<"ADMIN" | "SUPER_ADMIN">("ADMIN");
   415→  const [requirePasswordChange, setRequirePasswordChange] = useState(true);
   416→  const [saving, setSaving] = useState(false);
   417→
   418→  function reset() {
   419→    setEmail("");
   420→    setName("");
   421→    setPassword("");
   422→    setRole("ADMIN");
   423→    setRequirePasswordChange(true);
   424→  }
   425→
   426→  async function handleCreate() {
   427→    if (!email || !password) {
   428→      toast.error("Email and password are required");
   429→      return;
   430→    }
   431→    if (password.length < 8) {
   432→      toast.error("Password must be at least 8 characters");
   433→      return;
   434→    }
   435→    setSaving(true);
   436→    try {
   437→      const res = await fetch("/api/admin/users", {
   438→        method: "POST",
   439→        headers: { "Content-Type": "application/json" },
   440→        body: JSON.stringify({
   441→          email: email.trim().toLowerCase(),
   442→          name: name.trim() || undefined,
   443→          password,
   444→          role,
   445→          passwordChangeRequired: requirePasswordChange,
   446→        }),
   447→      });
   448→      if (!res.ok) {
   449→        const data = await res.json().catch(() => ({}));
   450→        throw new Error(data.error || "Create failed");
   451→      }
   452→      toast.success(`Created ${role} account for ${email}`);
   453→      reset();
   454→      onCreated();
   455→    } catch (err) {
   456→      toast.error(err instanceof Error ? err.message : "Create failed");
   457→    } finally {
   458→      setSaving(false);
   459→    }
   460→  }
   461→
   462→  return (
   463→    <Dialog open={open} onOpenChange={onOpenChange}>
   464→      <DialogContent>
   465→        <DialogHeader>
   466→          <DialogTitle>Add Admin User</DialogTitle>
   467→          <DialogDescription>
   468→            Create a new admin account. The user can sign in immediately with
   469→            the temporary password you set.
   470→          </DialogDescription>
   471→        </DialogHeader>
   472→        <div className="space-y-4 py-2">
   473→          <div className="space-y-2">
   474→            <Label htmlFor="cu-email">Email</Label>
   475→            <Input
   476→              id="cu-email"
   477→              type="email"
   478→              value={email}
   479→              onChange={(e) => setEmail(e.target.value)}
   480→              placeholder="name@mirdc.dost.gov.ph"
   481→              autoComplete="off"
   482→            />
   483→          </div>
   484→          <div className="space-y-2">
   485→            <Label htmlFor="cu-name">Display name (optional)</Label>
   486→            <Input
   487→              id="cu-name"
   488→              value={name}
   489→              onChange={(e) => setName(e.target.value)}
   490→              placeholder="Juan Dela Cruz"
   491→            />
   492→          </div>
   493→          <div className="space-y-2">
   494→            <Label htmlFor="cu-pass">Temporary password</Label>
   495→            <Input
   496→              id="cu-pass"
   497→              type="password"
   498→              value={password}
   499→              onChange={(e) => setPassword(e.target.value)}
   500→              placeholder="At least 8 characters"
   501→              autoComplete="new-password"
   502→            />
   503→            <p className="text-xs text-muted-foreground">
   504→              Communicate this securely to the user.
   505→            </p>
   506→          </div>
   507→          <div className="space-y-2">
   508→            <Label>Role</Label>
   509→            <Select
   510→              value={role}
   511→              onValueChange={(v: "ADMIN" | "SUPER_ADMIN") => setRole(v)}
   512→            >
   513→              <SelectTrigger>
   514→                <SelectValue />
   515→              </SelectTrigger>
   516→              <SelectContent>
   517→                <SelectItem value="ADMIN">Admin — manage chemicals & SDS</SelectItem>
   518→                <SelectItem value="SUPER_ADMIN">
   519→                  Super Admin — full access + user management
   520→                </SelectItem>
   521→              </SelectContent>
   522→            </Select>
   523→          </div>
   524→          <label
   525→            className="flex cursor-pointer items-start gap-3 rounded-md border p-3 hover:bg-muted/30"
   526→            htmlFor="cu-force-pw"
   527→          >
   528→            <input
   529→              id="cu-force-pw"
   530→              type="checkbox"
   531→              checked={requirePasswordChange}
   532→              onChange={(e) => setRequirePasswordChange(e.target.checked)}
   533→              className="mt-0.5 h-4 w-4 rounded border-input"
   534→            />
   535→            <div className="space-y-0.5">
   536→              <div className="text-sm font-medium">Require password change on next login</div>
   537→              <div className="text-xs text-muted-foreground">
   538→                User will be redirected to a password-change page before they
   539→                can reach the dashboard. Recommended for temporary passwords.
   540→              </div>
   541→            </div>
   542→          </label>
   543→        </div>
   544→        <DialogFooter>
   545→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   546→            Cancel
   547→          </Button>
   548→          <Button onClick={handleCreate} disabled={saving} className="gap-2">
   549→            {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   550→            Create User
   551→          </Button>
   552→        </DialogFooter>
   553→      </DialogContent>
   554→    </Dialog>
   555→  );
   556→}

   561→
   562→function EditUserDialog({
   563→  target,
   564→  onOpenChange,
   565→  onSaved,
   566→}: {
   567→  target: AdminUser | null;
   568→  onOpenChange: (open: boolean) => void;
   569→  onSaved: () => void;
   570→}) {
   571→  const { data: session } = useSession();
   572→  const isSelf = target?.id === session?.user?.id;
   573→
   574→  const [name, setName] = useState("");
   575→  const [role, setRole] = useState<"ADMIN" | "SUPER_ADMIN">("ADMIN");
   576→  const [disabled, setDisabled] = useState(false);
   577→  const [newPassword, setNewPassword] = useState("");
   578→  const [requirePwChange, setRequirePwChange] = useState(false);
   579→  const [saving, setSaving] = useState(false);
   580→
   581→  // Sync form state when target changes.
   582→  useEffect(() => {
   583→    if (target) {
   584→      setName(target.name ?? "");
   585→      setRole(target.role === "SUPER_ADMIN" ? "SUPER_ADMIN" : "ADMIN");
   586→      setDisabled(target.disabled);
   587→      setRequirePwChange(target.passwordChangeRequired);
   588→      setNewPassword("");
   589→    }
   590→  }, [target]);
   591→
   592→  async function handleSave() {
   593→    if (!target) return;
   594→    setSaving(true);
   595→    try {
   596→      const patch: Record<string, unknown> = {
   597→        name: name.trim() || null,
   598→        role,
   599→        disabled,
   600→        passwordChangeRequired: requirePwChange,
   601→      };
   602→      if (newPassword) {
   603→        if (newPassword.length < 8) {
   604→          toast.error("New password must be at least 8 characters");
   605→          setSaving(false);
   606→          return;
   607→        }
   608→        patch.password = newPassword;
   609→      }
   610→      const res = await fetch(`/api/admin/users/${target.id}`, {
   611→        method: "PATCH",
   612→        headers: { "Content-Type": "application/json" },
   613→        body: JSON.stringify(patch),
   614→      });
