    65→  formula: string;

    88→
    89→export function ChemicalManager() {
    90→  const [chemicals, setChemicals] = useState<AdminChemical[]>([]);
    91→  const [loading, setLoading] = useState(true);
    92→  const [search, setSearch] = useState("");
    93→  const [editing, setEditing] = useState<AdminChemical | null>(null);
    94→  const [creating, setCreating] = useState(false);
    95→  const [deleteTarget, setDeleteTarget] = useState<AdminChemical | null>(null);
    96→
    97→  const fetchChemicals = useCallback(async () => {
    98→    setLoading(true);
    99→    try {
   100→      const res = await fetch("/api/admin/chemicals");
   101→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   102→      const json = await res.json();
   103→      setChemicals(json.chemicals);
   104→    } catch (err) {
   105→      console.error("Failed to load chemicals:", err);
   106→    } finally {
   107→      setLoading(false);
   108→    }
   109→  }, []);
   110→
   111→  useEffect(() => {
   112→    fetchChemicals();
   113→  }, [fetchChemicals]);
   114→
   115→  const filtered = chemicals.filter((c) => {
   116→    const term = search.toLowerCase().trim();
   117→    if (!term) return true;
   118→    return (
   119→      c.chemicalName.toLowerCase().includes(term) ||
   120→      c.casNumber.toLowerCase().includes(term) ||
   121→      c.formula.toLowerCase().includes(term)
   122→    );
   123→  });
   124→
   125→  return (
   126→    <div className="space-y-4">
   127→      {/* Toolbar */}
   128→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   129→        <div className="relative flex-1">
   130→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   131→          <Input
   132→            value={search}
   133→            onChange={(e) => setSearch(e.target.value)}
   134→            placeholder="Search by name, CAS, or formula…"
   135→            className="h-10 pl-9"
   136→          />
   137→        </div>
   138→        <Button onClick={() => setCreating(true)} className="gap-2">
   139→          <Plus className="h-4 w-4" /> Add Chemical
   140→        </Button>
   141→        <Button onClick={fetchChemicals} variant="outline" size="icon" className="h-10 w-10">
   142→          <RefreshCw className="h-4 w-4" />
   143→        </Button>
   144→      </div>
   145→
   146→      {/* Table */}
   147→      <Card>
   148→        <CardContent className="p-0">
   149→          <div className="overflow-x-auto">
   150→            <table className="w-full text-sm">
   151→              <thead className="border-b bg-muted/30">
   152→                <tr className="text-left text-xs text-muted-foreground">
   153→                  <th className="px-4 py-3 font-medium">Chemical</th>
   154→                  <th className="px-4 py-3 font-medium">CAS</th>
   155→                  <th className="px-4 py-3 font-medium">Signal</th>
   156→                  <th className="px-4 py-3 font-medium">SDS</th>
   157→                  <th className="px-4 py-3 font-medium">Last Updated</th>
   158→                  <th className="px-4 py-3 font-medium text-right">Actions</th>
   159→                </tr>
   160→              </thead>
   161→              <tbody>
   162→                {loading ? (
   163→                  <tr>
   164→                    <td colSpan={6} className="py-12 text-center text-muted-foreground">
   165→                      <Loader2 className="mx-auto h-6 w-6 animate-spin text-teal-600" />
   166→                    </td>
   167→                  </tr>
   168→                ) : filtered.length === 0 ? (
   169→                  <tr>
   170→                    <td colSpan={6} className="py-12 text-center text-muted-foreground">
   171→                      No chemicals found.
   172→                    </td>
   173→                  </tr>
   174→                ) : (
   175→                  filtered.map((c) => (
   176→                    <tr key={c.id} className="border-b last:border-0 hover:bg-muted/20">
   177→                      <td className="px-4 py-3">
   178→                        <div className="font-medium">{c.chemicalName}</div>
   179→                        <div className="text-xs text-muted-foreground">{c.formula}</div>
   180→                      </td>
   181→                      <td className="px-4 py-3 font-mono text-xs">{c.casNumber}</td>
   182→                      <td className="px-4 py-3">
   183→                        <Badge
   184→                          className={cn(
   185→                            "text-[10px] uppercase",
   186→                            c.signalWord === "danger"
   187→                              ? "border-red-300 bg-red-50 text-red-700 dark:border-red-800 dark:bg-red-950 dark:text-red-300"
   188→                              : "border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-300"
   189→                          )}
   190→                        >
   191→                          {c.signalWord}
   192→                        </Badge>
   193→                      </td>
   194→                      <td className="px-4 py-3">
   195→                        {c.sds?.status === "available" ? (
   196→                          <Badge className="border-emerald-300 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">Available</Badge>
   197→                        ) : (
   198→                          <Badge className="border-amber-300 bg-amber-50 text-amber-700 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-300">Placeholder</Badge>
   199→                        )}
   200→                      </td>
   201→                      <td className="px-4 py-3 text-xs text-muted-foreground">
   202→                        {new Date(c.lastUpdated).toLocaleDateString("en-PH", { year: "numeric", month: "short", day: "numeric" })}
   203→                      </td>
   204→                      <td className="px-4 py-3">
   205→                        <div className="flex items-center justify-end gap-1">
   206→                          <Button size="icon" variant="ghost" className="h-8 w-8" onClick={() => setEditing(c)}>
   207→                            <Pencil className="h-4 w-4" />
   208→                          </Button>
   209→                          <Button size="icon" variant="ghost" className="h-8 w-8 text-red-600 hover:bg-red-50" onClick={() => setDeleteTarget(c)}>
   210→                            <Trash2 className="h-4 w-4" />
   211→                          </Button>
   212→                        </div>
   213→                      </td>
   214→                    </tr>
   215→                  ))
   216→                )}
   217→              </tbody>
   218→            </table>
   219→          </div>
   220→        </CardContent>
   221→      </Card>
   222→
   223→      {/* Create / Edit dialog */}
   224→      {(creating || editing) && (
   225→        <ChemicalFormDialog
   226→          chemical={editing}
   227→          onClose={() => {
   228→            setCreating(false);
   229→            setEditing(null);
   230→          }}
   231→          onSaved={() => {
   232→            setCreating(false);
   233→            setEditing(null);
   234→            fetchChemicals();
   235→          }}
   236→        />
   237→      )}
   238→
   239→      {/* Delete confirmation */}
   240→      <AlertDialog open={!!deleteTarget} onOpenChange={(open) => !open && setDeleteTarget(null)}>
   241→        <AlertDialogContent>
   242→          <AlertDialogHeader>
   243→            <AlertDialogTitle>Delete {deleteTarget?.chemicalName}?</AlertDialogTitle>
   244→            <AlertDialogDescription>
   245→              This will soft-delete the chemical and its SDS. The change will propagate to all synced devices. This action can be undone by re-adding the chemical.
   246→            </AlertDialogDescription>
   247→          </AlertDialogHeader>
   248→          <AlertDialogFooter>
   249→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   250→            <AlertDialogAction
   251→              className="bg-red-600 hover:bg-red-700"
   252→              onClick={async () => {
   253→                if (!deleteTarget) return;
   254→                try {
   255→                  const res = await fetch(`/api/admin/chemicals/${deleteTarget.id}`, { method: "DELETE" });
   256→                  if (!res.ok) throw new Error("Delete failed");
   257→                  setDeleteTarget(null);
   258→                  fetchChemicals();
   259→                } catch (err) {
   260→                  alert("Failed to delete chemical.");
   261→                }
   262→              }}
   263→            >
   264→              Delete
   265→            </AlertDialogAction>
   266→          </AlertDialogFooter>
   267→        </AlertDialogContent>
   268→      </AlertDialog>
   269→    </div>
   270→  );
   271→}

   276→
   277→interface FormState {
   278→  id: string;
   279→  casNumber: string;
   280→  chemicalName: string;
   281→  formula: string;
   282→  tradeName: string;
   283→  manufacturer: string;
   284→  supplier: string;
   285→  signalWord: SignalWord;
   286→  hazardClasses: HazardClass[];
   287→  ghsPictograms: GhsPictogram[];
   288→  storageLocation: string;
   289→  department: Department;
   290→  safetyInstructions: string;
   291→  version: string;
   292→  emergencyContact: string;
   293→  personalProtectiveEquipment: string;
   294→  regulatoryTags: string[];
   295→  firstAidMeasures: string;
   296→  firefightingMeasures: string;
   297→  accidentalReleaseMeasures: string;
   298→}
   299→
   300→function ChemicalFormDialog({
   301→  chemical,
   302→  onClose,
   303→  onSaved,
   304→}: {
   305→  chemical: AdminChemical | null;
   306→  onClose: () => void;
   307→  onSaved: () => void;
   308→}) {
   309→  const [form, setForm] = useState<FormState>({
   310→    id: chemical?.id ?? "",
   311→    casNumber: chemical?.casNumber ?? "",
   312→    chemicalName: chemical?.chemicalName ?? "",
   313→    formula: chemical?.formula ?? "",
   314→    tradeName: chemical?.tradeName ?? "",
   315→    manufacturer: chemical?.manufacturer ?? "",
   316→    supplier: chemical?.supplier ?? "",
   317→    signalWord: (chemical?.signalWord as SignalWord) ?? "danger",
   318→    hazardClasses: (chemical?.hazardClasses as HazardClass[]) ?? [],
   319→    ghsPictograms: (chemical?.ghsPictograms as GhsPictogram[]) ?? [],
   320→    storageLocation: chemical?.storageLocation ?? "",
   321→    department: (chemical?.department as Department) ?? "Chemical Analysis",
   322→    safetyInstructions: chemical?.safetyInstructions ?? "",
   323→    version: chemical?.version ?? "1.0",
   324→    emergencyContact: chemical?.emergencyContact ?? "",
   325→    personalProtectiveEquipment: chemical?.personalProtectiveEquipment.join("\n") ?? "",
   326→    regulatoryTags: chemical?.regulatoryTags ?? [],
   327→    firstAidMeasures: chemical?.firstAidMeasures ?? "",
   328→    firefightingMeasures: chemical?.firefightingMeasures ?? "",
   329→    accidentalReleaseMeasures: chemical?.accidentalReleaseMeasures ?? "",
   330→  });
   331→  const [saving, setSaving] = useState(false);
   332→  const [error, setError] = useState<string | null>(null);
   333→
   334→  // AI auto-fill state.
   335→  const [extracting, setExtracting] = useState(false);
   336→  const [extractError, setExtractError] = useState<string | null>(null);
   337→  const [extractedFromPdf, setExtractedFromPdf] = useState(false);
   338→  const fileInputRef = useRef<HTMLInputElement>(null);
   339→
   340→  const isEdit = !!chemical;
   341→
   342→  // ---------------------------------------------------------------------------
   343→  // AI auto-fill handler.
   344→  // Sends the selected PDF to /api/admin/sds/extract, then merges the returned
   345→  // fields into the form state. The `id` field is preserved (admin must enter
   346→  // it manually in create mode).
   347→  // ---------------------------------------------------------------------------
   348→  const handleAutoFill = async (e: React.ChangeEvent<HTMLInputElement>) => {
   349→    const file = e.target.files?.[0];
   350→    // Reset the input value so the same file can be re-selected later.
   351→    if (fileInputRef.current) fileInputRef.current.value = "";
   352→    if (!file) return;
   353→
   354→    setExtracting(true);
   355→    setExtractError(null);
   356→
   357→    try {
   358→      const fd = new FormData();
   359→      fd.append("file", file);
   360→
   361→      const res = await fetch("/api/admin/sds/extract", {
   362→        method: "POST",
   363→        body: fd,
   364→      });
   365→
   366→      const json = await res.json().catch(() => ({ success: false, error: "Invalid response from server" }));
   367→
   368→      if (!res.ok || !json.success) {
   369→        throw new Error(json.error || `Extraction failed (HTTP ${res.status})`);
   370→      }
   371→
   372→      const d = json.data as {
   373→        chemicalName?: string;
   374→        casNumber?: string;
   375→        formula?: string;
   376→        tradeName?: string;
   377→        manufacturer?: string;
   378→        supplier?: string;
   379→        signalWord?: SignalWord;
   380→        ghsPictograms?: GhsPictogram[];
   381→        hazardClasses?: HazardClass[];
   382→        storageLocation?: string;
   383→        safetyInstructions?: string;
   384→        emergencyContact?: string;
   385→        personalProtectiveEquipment?: string[];
   386→        firstAidMeasures?: string;
   387→        firefightingMeasures?: string;
   388→        accidentalReleaseMeasures?: string;
   389→      };
   390→
   391→      setForm((prev) => ({
   392→        ...prev,
   393→        // Preserve `id` — admin must always enter it manually in create mode.
   394→        id: prev.id,
   395→        chemicalName: d.chemicalName || prev.chemicalName,
   396→        casNumber: d.casNumber || prev.casNumber,
   397→        formula: d.formula || prev.formula,
   398→        tradeName: d.tradeName || prev.tradeName,
   399→        manufacturer: d.manufacturer || prev.manufacturer,
   400→        supplier: d.supplier || prev.supplier,
   401→        signalWord: d.signalWord || prev.signalWord,
   402→        ghsPictograms:
   403→          Array.isArray(d.ghsPictograms) && d.ghsPictograms.length > 0
   404→            ? d.ghsPictograms
   405→            : prev.ghsPictograms,
   406→        hazardClasses:
   407→          Array.isArray(d.hazardClasses) && d.hazardClasses.length > 0
   408→            ? d.hazardClasses
   409→            : prev.hazardClasses,
   410→        storageLocation: d.storageLocation || prev.storageLocation,
   411→        safetyInstructions: d.safetyInstructions || prev.safetyInstructions,
   412→        emergencyContact: d.emergencyContact || prev.emergencyContact,
   413→        // PPE comes back as an array; join with newlines for the textarea.
   414→        personalProtectiveEquipment:
   415→          Array.isArray(d.personalProtectiveEquipment) && d.personalProtectiveEquipment.length > 0
   416→            ? d.personalProtectiveEquipment.join("\n")
   417→            : prev.personalProtectiveEquipment,
   418→        firstAidMeasures: d.firstAidMeasures || prev.firstAidMeasures,
   419→        firefightingMeasures: d.firefightingMeasures || prev.firefightingMeasures,
   420→        accidentalReleaseMeasures:
   421→          d.accidentalReleaseMeasures || prev.accidentalReleaseMeasures,
   422→      }));
   423→
   424→      setExtractedFromPdf(true);
   425→    } catch (err) {
   426→      setExtractError(err instanceof Error ? err.message : "Auto-fill failed");
   427→      setExtractedFromPdf(false);
   428→    } finally {
   429→      setExtracting(false);
   430→    }
   431→  };
   432→
   433→  const handleSubmit = async (e: React.FormEvent) => {
   434→    e.preventDefault();
   435→    setSaving(true);
   436→    setError(null);
   437→
   438→    try {
   439→      const payload = {
   440→        ...form,
   441→        personalProtectiveEquipment: form.personalProtectiveEquipment
   442→          .split("\n")
   443→          .map((s) => s.trim())
   444→          .filter(Boolean),
   445→      };
   446→
   447→      const url = isEdit ? `/api/admin/chemicals/${form.id}` : "/api/admin/chemicals";
   448→      const method = isEdit ? "PUT" : "POST";
   449→
   450→      const res = await fetch(url, {
   451→        method,
   452→        headers: { "Content-Type": "application/json" },
   453→        body: JSON.stringify(payload),
   454→      });
   455→
   456→      if (!res.ok) {
   457→        const err = await res.json().catch(() => ({ error: "Request failed" }));
   458→        throw new Error(err.error || `HTTP ${res.status}`);
   459→      }
   460→
   461→      onSaved();
   462→    } catch (err) {
   463→      setError(err instanceof Error ? err.message : "Save failed");
   464→    } finally {
   465→      setSaving(false);
   466→    }
   467→  };
   468→
   469→  const toggleArray = <T,>(arr: T[], value: T): T[] => {
   470→    return arr.includes(value) ? arr.filter((x) => x !== value) : [...arr, value];
   471→  };
   472→
   473→  return (
   474→    <Dialog open={true} onOpenChange={(open) => !open && onClose()}>
   475→      <DialogContent className="max-h-[90vh] max-w-2xl overflow-y-auto">
   476→        <DialogHeader>
   477→          <DialogTitle>{isEdit ? "Edit Chemical" : "Add New Chemical"}</DialogTitle>
   478→          <DialogDescription>
   479→            {isEdit
   480→              ? "Update the chemical record. Changes will sync to all devices."
   481→              : "Create a new chemical. A placeholder SDS will be generated automatically."}
   482→          </DialogDescription>
   483→        </DialogHeader>
   484→
   485→        {/* Hidden file input — triggered by the Auto-fill button. */}
   486→        <input
   487→          ref={fileInputRef}
   488→          type="file"
   489→          accept=".pdf,application/pdf"
   490→          hidden
   491→          onChange={handleAutoFill}
   492→        />
   493→
   494→        <form onSubmit={handleSubmit} className="space-y-4">
   495→          {/* Auto-fill from PDF banner / button */}
   496→          <div className="space-y-2 rounded-lg border border-teal-200 bg-teal-50/60 p-3 dark:border-teal-900 dark:bg-teal-950/40">
   497→            <div className="flex flex-wrap items-center gap-2">
   498→              <Button
   499→                type="button"
   500→                variant="outline"
   501→                size="sm"
   502→                className="gap-2 border-teal-400 text-teal-700 hover:bg-teal-100 dark:border-teal-800 dark:text-teal-300 dark:hover:bg-teal-900/40"
   503→                onClick={() => fileInputRef.current?.click()}
   504→                disabled={extracting || saving}
   505→              >
   506→                {extracting ? (
   507→                  <Loader2 className="h-4 w-4 animate-spin" />
   508→                ) : (
   509→                  <FileText className="h-4 w-4" />
   510→                )}
   511→                {extracting ? "Reading SDS document…" : "Auto-fill from PDF"}
   512→              </Button>
   513→              <span className="text-xs text-muted-foreground">
   514→                Upload a scanned or digital SDS PDF — AI will read it and pre-fill the fields below.
   515→              </span>
   516→            </div>
   517→
   518→            {/* Loading state */}
   519→            {extracting && (
   520→              <div className="flex items-center gap-2 text-xs text-teal-700 dark:text-teal-300">
   521→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   522→                Converting PDF to images, then reading with AI… This takes ~10-15 seconds for multi-page documents.
   523→              </div>
   524→            )}
   525→
   526→            {/* Error banner */}
   527→            {extractError && !extracting && (
   528→              <div className="flex items-start gap-2 rounded-md border border-amber-300 bg-amber-50 px-3 py-2 text-xs text-amber-800 dark:border-amber-800 dark:bg-amber-950 dark:text-amber-200">
   529→                <AlertCircle className="mt-0.5 h-4 w-4 shrink-0" />
   530→                <div className="flex-1">
   531→                  <div className="font-semibold">Auto-fill failed</div>
   532→                  <div className="mt-0.5">{extractError}</div>
   533→                </div>
   534→                <button
   535→                  type="button"
   536→                  className="shrink-0 rounded p-0.5 hover:bg-amber-100 dark:hover:bg-amber-900"
   537→                  onClick={() => setExtractError(null)}
   538→                  aria-label="Dismiss error"
   539→                >
   540→                  <X className="h-3.5 w-3.5" />
   541→                </button>
   542→              </div>
   543→            )}
   544→
   545→            {/* Review banner */}
   546→            {extractedFromPdf && !extracting && (
   547→              <div className="flex items-start gap-2 rounded-md border border-teal-300 bg-teal-100/70 px-3 py-2 text-xs text-teal-800 dark:border-teal-700 dark:bg-teal-900/50 dark:text-teal-200">
   548→                <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0" />
   549→                <div className="flex-1">
   550→                  <div className="font-semibold">Auto-filled from PDF</div>
   551→                  <div className="mt-0.5">Please review all fields carefully before saving — AI extraction may have errors or omissions.</div>
   552→                </div>
   553→                <button
   554→                  type="button"
   555→                  className="shrink-0 rounded p-0.5 hover:bg-teal-200 dark:hover:bg-teal-800"
   556→                  onClick={() => setExtractedFromPdf(false)}
   557→                  aria-label="Dismiss review banner"
   558→                >
